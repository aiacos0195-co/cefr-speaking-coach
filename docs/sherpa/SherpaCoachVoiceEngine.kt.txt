package com.example.cefrspeakingcoach

import android.content.Context
import android.os.Handler
import android.os.Looper
import android.util.Log
import com.k2fsa.sherpa.onnx.OfflineTts
import com.k2fsa.sherpa.onnx.OfflineTtsConfig
import com.k2fsa.sherpa.onnx.OfflineTtsModelConfig
import com.k2fsa.sherpa.onnx.OfflineTtsVitsModelConfig
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.ensureActive
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import java.io.File
import java.util.concurrent.atomic.AtomicInteger

/**
 * Offline neural TTS backed by sherpa-onnx running a Piper VITS model.
 *
 * Design notes worth keeping in mind:
 *
 * - The model is loaded from filesDir, not from assets. Loading straight from
 *   assets works only if aapt leaves the .onnx uncompressed, and it forces a
 *   60-80 MB model into the APK. Installing to filesDir keeps the download
 *   optional and the APK small. See SherpaModelInstaller.
 *
 * - All six coaches share ONE model file. A multi-speaker Piper model exposes
 *   hundreds of speaker ids; each coach is just a different sid. Six separate
 *   voice packs would be six times the disk for no benefit.
 *
 * - ConversationVoiceModel.pitch is ignored. VITS has no pitch parameter, and
 *   pitch-shifting the output after the fact sounds worse than picking a
 *   different sid. Distinguish coaches by speaker id, not by pitch.
 *
 * - Synthesis is not real time on older phones. A long sentence on a low-end
 *   device can take a second or more, which is why speak() returns immediately
 *   and completion arrives through the callbacks.
 */
class SherpaCoachVoiceEngine(
    context: Context,
    private val callbacks: CoachVoiceEngineCallbacks = CoachVoiceEngineCallbacks(),
    private val speakerIds: Map<String, Int> = DEFAULT_SPEAKER_IDS
) : CoachVoiceEngine, CoachAwareVoiceEngine {

    override val engineId: String = "sherpa_onnx"
    override val displayName: String = "Sherpa-ONNX Neural Voice"

    private val appContext = context.applicationContext
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Main)
    private val mainHandler = Handler(Looper.getMainLooper())
    private val player = PcmCoachAudioPlayer()
    private val installer = SherpaModelInstaller(appContext)

    private val generation = AtomicInteger(0)

    @Volatile
    private var tts: OfflineTts? = null

    @Volatile
    private var speakJob: Job? = null

    @Volatile
    private var ready: Boolean = false

    @Volatile
    private var availableSpeakers: Int = 0

    /** Last failure reason, so a settings screen can show something useful. */
    var lastError: String? = null
        private set

    override fun initialize() {
        if (ready) return

        scope.launch {
            val loaded = withContext(Dispatchers.IO) {
                runCatching { loadModel() }
            }

            loaded
                .onSuccess { engine ->
                    tts = engine
                    availableSpeakers = runCatching { engine.numSpeakers() }.getOrDefault(1)
                    ready = true
                    lastError = null

                    Log.i(
                        TAG,
                        "sherpa-onnx ready: ${availableSpeakers} speaker(s) at " +
                            "${runCatching { engine.sampleRate() }.getOrDefault(0)} Hz"
                    )

                    postReady()
                }
                .onFailure { error ->
                    ready = false
                    lastError = error.message ?: "Model load failed"
                    Log.w(TAG, "sherpa-onnx unavailable: $lastError")
                    // Deliberately quiet: an uninstalled model is the normal
                    // state before first download, not an error worth showing.
                }
        }
    }

    private fun loadModel(): OfflineTts {
        val modelDir = installer.installedModelDir()
            ?: throw IllegalStateException("No sherpa model installed")

        val modelFile = installer.findModelFile(modelDir)
            ?: throw IllegalStateException("No .onnx file in ${modelDir.absolutePath}")

        val tokensFile = File(modelDir, "tokens.txt")
        require(tokensFile.exists()) { "Missing tokens.txt in ${modelDir.absolutePath}" }

        val dataDir = File(modelDir, "espeak-ng-data")
        require(dataDir.isDirectory) { "Missing espeak-ng-data in ${modelDir.absolutePath}" }

        val config = OfflineTtsConfig(
            model = OfflineTtsModelConfig(
                vits = OfflineTtsVitsModelConfig(
                    model = modelFile.absolutePath,
                    tokens = tokensFile.absolutePath,
                    dataDir = dataDir.absolutePath,
                    lexicon = ""
                ),
                numThreads = 2,
                debug = false,
                provider = "cpu"
            ),
            // Split long replies into sentences so the first audio arrives sooner.
            maxNumSentences = 1
        )

        return OfflineTts(config = config)
    }

    override fun isReady(): Boolean {
        return ready && tts != null
    }

    override fun canSpeakAs(voice: ConversationVoiceModel): Boolean {
        if (!isReady()) return false

        val sid = speakerIds[voice.id.lowercase()] ?: return false
        return sid < availableSpeakers
    }

    override fun speak(
        text: String,
        voice: ConversationVoiceModel
    ): Boolean {
        val engine = tts ?: return false
        if (!ready || text.isBlank()) return false

        val sid = speakerIds[voice.id.lowercase()] ?: return false
        if (sid >= availableSpeakers) {
            Log.w(TAG, "Speaker id $sid out of range (model has $availableSpeakers)")
            return false
        }

        cancelCurrentSpeech()
        val myGeneration = generation.incrementAndGet()

        speakJob = scope.launch {
            try {
                postStarted()

                val generated = withContext(Dispatchers.IO) {
                    engine.generate(
                        text = sanitize(text),
                        sid = sid,
                        speed = voice.speechRate.coerceIn(0.5f, 2.0f)
                    )
                }

                ensureActive()
                if (generation.get() != myGeneration) return@launch

                if (generated.samples.isEmpty()) {
                    postError("Sherpa produced no audio")
                    return@launch
                }

                player.play(
                    audio = CoachVoiceAudio(
                        samples = generated.samples,
                        sampleRate = generated.sampleRate,
                        channels = 1
                    ),
                    onCompleted = {
                        if (generation.get() == myGeneration) postCompleted()
                    },
                    onError = { message ->
                        if (generation.get() == myGeneration) postError(message)
                    }
                )
            } catch (e: Exception) {
                if (generation.get() == myGeneration) {
                    Log.e(TAG, "Synthesis failed", e)
                    postError(e.message ?: "Sherpa synthesis failed")
                }
            }
        }

        return true
    }

    override fun stop() {
        cancelCurrentSpeech()
        player.stop()
    }

    override fun release() {
        ready = false
        cancelCurrentSpeech()
        player.release()

        runCatching { tts?.release() }
        tts = null

        runCatching { scope.cancel() }
    }

    private fun cancelCurrentSpeech() {
        generation.incrementAndGet()
        speakJob?.cancel()
        speakJob = null
    }

    /**
     * espeak-ng handles most punctuation, but stray markdown and repeated
     * whitespace from the coach engine make it stumble audibly.
     */
    private fun sanitize(text: String): String {
        return text
            .replace(Regex("[*_`#>]"), " ")
            .replace(Regex("\\s+"), " ")
            .trim()
    }

    private fun postReady() {
        mainHandler.post { callbacks.onReady(engineId) }
    }

    private fun postStarted() {
        mainHandler.post { callbacks.onStarted(engineId) }
    }

    private fun postCompleted() {
        mainHandler.post { callbacks.onCompleted(engineId) }
    }

    private fun postError(message: String) {
        mainHandler.post { callbacks.onError(engineId, message) }
    }

    companion object {
        private const val TAG = "SherpaCoachVoiceEngine"

        /**
         * PLACEHOLDER speaker ids.
         *
         * These are not tuned. With a multi-speaker model such as
         * vits-piper-en_US-libritts_r-medium you get hundreds of voices, and
         * which sid sounds like "a calm British man" is something you have to
         * listen to. Use the audition screen described in
         * SHERPA_INTEGRATION.md, pick six, then replace these numbers.
         *
         * With a single-speaker model every entry must be 0, and all six
         * coaches will sound identical. That is expected, not a bug.
         */
        val DEFAULT_SPEAKER_IDS: Map<String, Int> = mapOf(
            "sophie" to 0,
            "emma" to 1,
            "lily" to 2,
            "james" to 3,
            "ethan" to 4,
            "luca" to 5
        )
    }
}
